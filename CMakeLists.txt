cmake_minimum_required(VERSION 3.25)
project(entities_unity_roundtable C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(WAMR_ROOT "" CACHE PATH "wasm-micro-runtime checkout (commit in UPSTREAM)")
set(LIBRISCV_ROOT "" CACHE PATH "libriscv checkout (commit in UPSTREAM)")
if(NOT WAMR_ROOT OR NOT LIBRISCV_ROOT)
	message(FATAL_ERROR "set -DWAMR_ROOT=... -DLIBRISCV_ROOT=... (tools/build.exs does)")
endif()
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Release)
endif()

# WAMR in-process, as the Unity plugin would carry it: interpreter (fast) and, when WAMR_AOT=1, AOT modules too.
set(WAMR_BUILD_PLATFORM ${CMAKE_SYSTEM_NAME})
string(TOLOWER "${WAMR_BUILD_PLATFORM}" WAMR_BUILD_PLATFORM)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
	set(WAMR_BUILD_TARGET AARCH64)
else()
	set(WAMR_BUILD_TARGET X86_64)
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 1)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 0)
include(${WAMR_ROOT}/build-scripts/runtime_lib.cmake)
add_library(vmlib STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_compile_options(vmlib PRIVATE -w)

set(RISCV_64I ON CACHE BOOL "" FORCE)
set(RISCV_EXT_C ON CACHE BOOL "" FORCE)
set(RISCV_EXT_A ON CACHE BOOL "" FORCE)
set(RISCV_EXT_F ON CACHE BOOL "" FORCE)
set(RISCV_EXT_D ON CACHE BOOL "" FORCE)
set(RISCV_BINARY_TRANSLATION OFF CACHE BOOL "" FORCE)
add_subdirectory(${LIBRISCV_ROOT}/lib libriscv)

# The vm library: both runtimes behind one C API (Unity P/Invokes this).
add_library(rtbench SHARED native/rt.cpp)
target_include_directories(rtbench PRIVATE ${WAMR_ROOT}/core/iwasm/include)
target_link_libraries(rtbench PRIVATE vmlib riscv)
set_target_properties(rtbench PROPERTIES CXX_VISIBILITY_PRESET default)

# The control: bench.c compiled for the host, called directly.
add_library(bench_native STATIC guests/bench.c)
target_compile_options(bench_native PRIVATE -O2 -ffp-contract=off -fno-math-errno)
add_executable(rtbench_cli native/main.cpp)
target_link_libraries(rtbench_cli PRIVATE rtbench bench_native)
